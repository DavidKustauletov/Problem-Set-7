Содержимое log.sql:
-- 1. Узнаем детали происшествия за 28 июля 2025 года на Humphrey Street
SELECT description 
FROM crime_scene_reports 
WHERE year = 2025 AND month = 7 AND day = 28 AND street = 'Humphrey Street';

-- 2. Смотрим показания свидетелей, упоминающих пекарню (bakery)
SELECT name, transcript 
FROM interviews 
WHERE year = 2025 AND month = 7 AND day = 28 AND transcript LIKE '%bakery%';

-- 3. Находим номера машин, уехавших с парковки пекарни в течение 10 минут после кражи (10:15 - 10:25)
SELECT license_plate 
FROM bakery_security_logs 
WHERE year = 2025 AND month = 7 AND day = 28 AND hour = 10 AND minute BETWEEN 15 AND 25 AND activity = 'exit';

-- 4. Находим номера счетов людей, снимавших наличные в банкомате на Leggett Street утром в день кражи
SELECT account_number 
FROM atm_transactions 
WHERE year = 2025 AND month = 7 AND day = 28 AND atm_location = 'Leggett Street' AND transaction_type = 'withdraw';

-- 5. Ищем короткие звонки (меньше минуты) в день кражи
SELECT caller, receiver 
FROM phone_calls 
WHERE year = 2025 AND month = 7 AND day = 28 AND duration < 60;

-- 6. Находим самый ранний рейс из Fiftyville на следующий день (29 июля 2025)
SELECT flights.id, destination_airport_id, hour, minute 
FROM flights 
JOIN airports ON flights.origin_airport_id = airports.id 
WHERE airports.city = 'Fiftyville' AND flights.year = 2025 AND flights.month = 7 AND flights.day = 29 
ORDER BY flights.hour ASC, flights.minute ASC 
LIMIT 1;

-- 7. Сопоставляем все улики вместе, чтобы найти вора (Bruce)
SELECT name 
FROM people 
WHERE phone_number IN (
    SELECT caller FROM phone_calls WHERE year = 2025 AND month = 7 AND day = 28 AND duration < 60
)
AND license_plate IN (
    SELECT license_plate FROM bakery_security_logs WHERE year = 2025 AND month = 7 AND day = 28 AND hour = 10 AND minute BETWEEN 15 AND 25
)
AND bank_account_number IN (
    SELECT account_number FROM bank_accounts WHERE account_number IN (
        SELECT account_number FROM atm_transactions WHERE year = 2025 AND month = 7 AND day = 28 AND atm_location = 'Leggett Street' AND transaction_type = 'withdraw'
    )
)
AND passport_number IN (
    SELECT passport_number FROM passengers WHERE flight_id = (
        SELECT flights.id FROM flights 
        JOIN airports ON flights.origin_airport_id = airports.id 
        WHERE airports.city = 'Fiftyville' AND flights.year = 2025 AND flights.month = 7 AND flights.day = 29 
        ORDER BY flights.hour ASC, flights.minute ASC LIMIT 1
    )
);

-- 8. Находим город назначения самого раннего рейса (New York City)
SELECT city 
FROM airports 
WHERE id = (
    SELECT destination_airport_id FROM flights 
    JOIN airports ON flights.origin_airport_id = airports.id 
    WHERE airports.city = 'Fiftyville' AND flights.year = 2025 AND flights.month = 7 AND flights.day = 29 
    ORDER BY flights.hour ASC, flights.minute ASC LIMIT 1
);

-- 9. Находим сообщника по номеру телефона, на который звонил вор (Robin)
SELECT name 
FROM people 
WHERE phone_number = (
    SELECT receiver FROM phone_calls 
    WHERE year = 2025 AND month = 7 AND day = 28 AND duration < 60 
    AND caller = (SELECT phone_number FROM people WHERE name = 'Bruce')
);
